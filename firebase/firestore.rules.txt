rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read, write: if request.auth != null;
    }
    match /tickets/{ticket} {
      allow read: if true;
      allow write: if false;
    }
    match /composers/{composer} {
      allow read: if true;
      allow write: if false;
    }
    match /tests/{test} {
      allow read: if true;
      allow write: if false;
    }
  }
}
